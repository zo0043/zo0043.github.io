{{- /* 指南条目闸门：条目格式机检（全站只跑一次，由 layouts/escape/list.html 调用）。 为什么用 errorf 而不是外挂校验脚本：本站已有的双链层就是构建期硬失败 （标题/别名撞车走 errorf），条目格式同理——坏条目不该被部署出去。 脚本方案要在 CI 里额外跑一遍且可能忘记跑，构建期检查无法绕过。 检查项（每条 errorf 都会让 hugo 以非 0 退出，但页面仍会渲染—— 所以判断成败只看退出码，不要看输出目录里有没有页面）： (1) description 必填（缺了会让 og/twitter/RSS 回退正文摘要） (2) chapter 必填（列表页按它分组） (3) cost.money / cost.time / cost.will 取值合法 (4) gain 取值合法 (5) metric 取值合法（口径：不同口径之间不比较） (6) evidence 取值合法 (7) checked 必填：来源的核对日期（规则会变，说法有保质期） (8) 正文五个小节齐全：怎么做 / 说人话 / 收益 / 来源 / 备注 (9) 「说人话」小节里不得出现阿拉伯数字——这一节只负责解释， 不许引入收益栏之外的新数字（上游 check-plain 的同款纪律） 兼容性：CI 固定 0.130.0。这里只做字符串比较，不碰 0.130 下会报错的 `strings.TrimSpace .` 之类写法；split 的结果是 []string，`. ` 有确切类型。 */ -}} {{- $validMoney := slice "0" "少" "多" -}} {{- $validTime := slice "少" "中" "多" -}} {{- $validWill := slice "否" "些" "是" -}} {{- $validGain := slice "大" "中" "小" -}} {{- $validMetric := slice "亏损" "收益" "时间" "自由" "处罚" -}} {{- $validEvidence := slice "A" "B" "C" -}} {{- $requiredSections := slice "怎么做" "说人话" "收益" "来源" "备注" -}} {{- range where site.RegularPages "Section" "escape" -}} {{- $ref := .RelPermalink -}} {{- $title := .Title -}} {{- if eq (trim (.Description | default "") " ") "" -}} {{- errorf "指南条目 %s（%s）缺 description：og/twitter/meta/RSS 会回退到正文摘要，把字面 [[…]] 泄漏出去。" $ref $title -}} {{- end -}} {{- if eq (printf "%v" (.Params.chapter | default "")) "" -}} {{- errorf "指南条目 %s（%s）缺 chapter：列表页按 chapter 分组，缺了会掉进「未归类」。" $ref $title -}} {{- end -}} {{- $cost := .Params.cost -}} {{- if not $cost -}} {{- errorf "指南条目 %s（%s）缺 cost（money/time/will）。" $ref $title -}} {{- else -}} {{- $m := printf "%v" (index $cost "money") -}} {{- $t := printf "%v" (index $cost "time") -}} {{- $w := printf "%v" (index $cost "will") -}} {{- if not (in $validMoney $m) -}}{{ errorf "指南条目 %s（%s）cost.money=%q 非法，可选：0/少/多。" $ref $title $m -}}{{- end -}} {{- if not (in $validTime $t) -}}{{ errorf "指南条目 %s（%s）cost.time=%q 非法，可选：少/中/多。" $ref $title $t -}}{{- end -}} {{- if not (in $validWill $w) -}}{{ errorf "指南条目 %s（%s）cost.will=%q 非法，可选：否/些/是。" $ref $title $w -}}{{- end -}} {{- end -}} {{- $gain := printf "%v" (.Params.gain | default "") -}} {{- if not (in $validGain $gain) -}}{{ errorf "指南条目 %s（%s）gain=%q 非法，可选：大/中/小。" $ref $title $gain -}}{{- end -}} {{- $metric := printf "%v" (.Params.metric | default "") -}} {{- if not (in $validMetric $metric) -}}{{ errorf "指南条目 %s（%s）metric=%q 非法，可选：亏损/收益/时间/自由/处罚。" $ref $title $metric -}}{{- end -}} {{- $ev := printf "%v" (.Params.evidence | default "") -}} {{- if not (in $validEvidence $ev) -}}{{ errorf "指南条目 %s（%s）evidence=%q 非法，可选：A/B/C。" $ref $title $ev -}}{{- end -}} {{- $checked := .Params.checked -}} {{- if not $checked -}} {{- errorf "指南条目 %s（%s）缺 checked（来源核对日期，形如 2026-10-08）。规则会变，没有核对日期的规则类说法无法判断是否过期。" $ref $title -}} {{- else -}} {{- $days := div (now.Sub (time.AsTime $checked)).Hours 24 -}} {{- if gt $days 180 -}} {{- warnf "指南条目 %s（%s）的来源已核对 %v 天，超过 180 天——规则可能已改，请重新核对。" $ref $title (int $days) -}} {{- end -}} {{- end -}} {{- /* 正文小节检查：按 "\n## " 切块，块首即为小节名 */ -}} {{- $chunks := split .RawContent "\n## " -}} {{- $found := slice -}} {{- $plain := "" -}} {{- range $chunks -}} {{- $head := trim . " \n\r" -}} {{- range $sec := $requiredSections -}} {{- if hasPrefix $head $sec -}} {{- $found = $found | append $sec -}} {{- if eq $sec "说人话" -}}{{- $plain = $head -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- range $sec := $requiredSections -}} {{- if not (in $found $sec) -}} {{- errorf "指南条目 %s（%s）缺小节「## %s」。五个小节必须齐全：怎么做/说人话/收益/来源/备注。" $ref $title $sec -}} {{- end -}} {{- end -}} {{- if and $plain (findRE "[0-9]" $plain) -}} {{- errorf "指南条目 %s（%s）的「说人话」小节里出现了阿拉伯数字。这一节只做解释，不许引入收益栏之外的新数字（把数字留在「收益」或「来源」里）。" $ref $title -}} {{- end -}} {{- end -}} {{- return true -}}
